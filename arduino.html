O que é o Arduíno?

Conceito e Funcionamento

O Arduíno é uma plataforma de prototipagem eletrônica de código aberto (open-source) baseada em hardware e software flexíveis e fáceis de usar. É constituído por um microcontrolador (o "cérebro"), portas de entrada e saída (pinos digitais e analógicos) e um ambiente de desenvolvimento integrado (IDE).

Seu funcionamento baseia-se em um ciclo infinito onde o microcontrolador lê sinais de sensores (entradas), processa essas informações com base no código programado (lógica de controle) e aciona atuadores (saídas, como LEDs, motores ou relés). Isso o torna perfeito para desenvolver projetos de Internet das Coisas e simular ambientes de Automação Industrial.

Exemplos de Programação

Como integrar e ler dados de nossos 10 sensores principais utilizando o Arduíno.

1. DHT11 (Temperatura e Umidade)

#include <DHT.h>
#define DHTPIN 2
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);
void setup() {
  Serial.begin(9600);
  dht.begin();
}
void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  Serial.print("Temp: "); Serial.println(t);
  delay(2000);
}

2. HC-SR04 (Ultrassônico/Distância)

#define trigPin 9
#define echoPin 10
void setup() {
  Serial.begin(9600);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
}
void loop() {
  digitalWrite(trigPin, LOW); delayMicroseconds(2);
  digitalWrite(trigPin, HIGH); delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  long duracao = pulseIn(echoPin, HIGH);
  long distancia = (duracao/2) / 29.1;
  Serial.println(distancia);
  delay(1000);
}

3. LDR (Luminosidade)

int pinoLDR = A0;
void setup() {
  Serial.begin(9600);
}
void loop() {
  int valorLuz = analogRead(pinoLDR);
  Serial.print("Luminosidade: ");
  Serial.println(valorLuz);
  delay(500);
}

4. PIR HC-SR501 (Movimento)

int pinoPIR = 7;
void setup() {
  pinMode(pinoPIR, INPUT);
  Serial.begin(9600);
}
void loop() {
  int movimento = digitalRead(pinoPIR);
  if (movimento == HIGH) {
    Serial.println("Movimento Detectado!");
  }
  delay(1000);
}

5. MQ-2 (Gás e Fumaça)

int pinoGas = A1;
void setup() {
  Serial.begin(9600);
}
void loop() {
  int nivelGas = analogRead(pinoGas);
  Serial.print("Nivel de Gas: ");
  Serial.println(nivelGas);
  if(nivelGas > 400) Serial.println("ALERTA!");
  delay(1000);
}

6. MFRC522 (RFID)

#include <MFRC522.h>

#define RST_PIN 9
MFRC522 mfrc522(SS_PIN, RST_PIN);
void setup() {
  Serial.begin(9600);
  SPI.begin();
  mfrc522.PCD_Init();
}
void loop() {
  if (mfrc522.PICC_IsNewCardPresent()) {
    // Ler ID do cartao...
  }
}

7. ACS712 (Corrente Elétrica)

int pinoCorrente = A2;
void setup() {
  Serial.begin(9600);
}
void loop() {
  int valor = analogRead(pinoCorrente);
  float tensao = (valor / 1024.0) * 5000; 
  float corrente = (tensao - 2500) / 66;
  Serial.print("Corrente (A): ");
  Serial.println(corrente);
  delay(1000);
}

8. Sensor de Chuva

9. KY-037 (Sensor de Som)

int pinoSom = A4;
int pinoDigital = 8;
void setup() {
  Serial.begin(9600);

}
void loop() {
  int nivelSom = analogRead(pinoSom);
  int detectado = digitalRead(pinoDigital);
  Serial.println(nivelSom);

}
